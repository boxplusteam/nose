const { spawn } = require('child_process');
const fs = require('fs');
const path = require('path');
const express = require('express');
const axios = require('axios');

const app = express();

const PORT = 8080;
const RAM_DISK = 'D:/HLS_LIVE'; // Tu disco RAM (súper rápido)
const JSON_URL = 'https://raw.githubusercontent.com/boxplusteam/nose/refs/heads/main/data.json';

// AJUSTES
const MAX_CANALES = 5;
const INACTIVIDAD_MS = 30000;

const canalesActivos = new Map();

// Crear carpeta en RAM Disk si no existe
if (!fs.existsSync(RAM_DISK)) {
    try {
        fs.mkdirSync(RAM_DISK, { recursive: true });
    } catch (e) {
        console.log("❌ ERROR: No se puede escribir en D:");
    }
}

function log(...msg) {
    console.log(new Date().toLocaleTimeString(), ...msg);
}

function eliminarCanal(idCanal) {
    const canal = canalesActivos.get(idCanal);
    if (!canal) return;

    log(`🛑 Cerrando canal: ${idCanal}`);
    if (canal.proceso) {
        canal.proceso.kill('SIGKILL'); // Muerte súbita al proceso para liberar memoria rápido
    }

    if (canal.timeout) clearTimeout(canal.timeout);

    // Limpieza de archivos tras un breve retraso
    setTimeout(() => {
        if (canal.folder && fs.existsSync(canal.folder)) {
            try {
                fs.rmSync(canal.folder, { recursive: true, force: true });
                log(`🗑️ Limpieza completada: ${idCanal}`);
            } catch (err) { /* Silencio */ }
        }
    }, 2000);

    canalesActivos.delete(idCanal);
}

function programarCierrePorInactividad(idCanal) {
    const canal = canalesActivos.get(idCanal);
    if (!canal) return;

    if (canal.timeout) clearTimeout(canal.timeout);

    canal.timeout = setTimeout(() => {
        const actual = canalesActivos.get(idCanal);
        if (actual && actual.usuarios.size === 0) {
            log(`⌛ Canal inactivo: ${idCanal} - Procediendo a cerrar`);
            eliminarCanal(idCanal);
        }
    }, INACTIVIDAD_MS);
}

async function iniciarCanal(idCanal) {
    if (canalesActivos.has(idCanal)) {
        return canalesActivos.get(idCanal);
    }

    if (canalesActivos.size >= MAX_CANALES) {
        for (const [id, canal] of canalesActivos) {
            if (canal.usuarios.size === 0) {
                eliminarCanal(id);
                break;
            }
        }
        if (canalesActivos.size >= MAX_CANALES) throw new Error('Servidor Saturado');
    }

    const { data } = await axios.get(JSON_URL);
    const canales = data.canales || data;
    const canalData = canales[idCanal];

    if (!canalData) throw new Error('Canal no encontrado');

    const folderPath = path.join(RAM_DISK, idCanal);
    if (!fs.existsSync(folderPath)) fs.mkdirSync(folderPath, { recursive: true });

    log(`🚀 PROCESANDO: ${idCanal} (Modo Passthrough - CPU 0%)`);

    // Argumentos ultraligeros: Solo reempaqueta, NO recodifica.
    const args = [
        '-reconnect', '1', '-reconnect_streamed', '1', '-reconnect_at_eof', '1',
        '-i', canalData.url,
        '-c', 'copy', // Esto abarca video (-c:v) y audio (-c:a) al mismo tiempo
        '-f', 'hls',
        '-hls_time', '4', // Tramos más cortos = carga más rápida en el reproductor
        '-hls_list_size', '4',
        '-hls_flags', 'delete_segments',
        path.join(folderPath, 'index.m3u8')
    ];

    const ffmpeg = spawn('ffmpeg', args);

    ffmpeg.stderr.on('data', (d) => {
        const output = d.toString();
        if (output.includes('Error')) console.log(`[FFMPEG ERR] ${idCanal}:`, output.substring(0, 60));
    });

    const canalObj = {
        id: idCanal,
        proceso: ffmpeg,
        folder: folderPath,
        usuarios: new Set(),
        timeout: null
    };

    canalesActivos.set(idCanal, canalObj);
    
    // Esperar 4 segundos (tiempo de un segmento) para asegurar que hay video antes de responder
    await new Promise(r => setTimeout(r, 4000));
    return canalObj;
}

// Optimización: CORS Global súper rápido
app.use((req, res, next) => {
    res.header('Access-Control-Allow-Origin', '*');
    next();
});

// Ruta para el Playlist (M3U8)
app.get('/:canal/index.m3u8', async (req, res) => {
    const idCanal = req.params.canal;
    const userId = req.headers['x-user-id'] || req.ip;

    try {
        let canal = canalesActivos.get(idCanal);
        if (!canal) {
            canal = await iniciarCanal(idCanal);
        }

        if (canal.timeout) {
            clearTimeout(canal.timeout);
            canal.timeout = null;
        }

        canal.usuarios.add(userId);
        
        req.on('close', () => {
            canal.usuarios.delete(userId);
            if (canal.usuarios.size === 0) programarCierrePorInactividad(idCanal);
        });

        const file = path.join(RAM_DISK, idCanal, 'index.m3u8');
        if (fs.existsSync(file)) {
            // EVITAR CACHÉ en M3U8 para que el reproductor siempre pida el más reciente
            res.setHeader('Cache-Control', 'no-cache, no-store, must-revalidate');
            return res.sendFile(file);
        }
        res.status(503).send("Cargando stream...");
    } catch (err) {
        res.status(500).send(err.message);
    }
});

// Ruta para los fragmentos de video (TS)
app.get('/:canal/:segmento.ts', (req, res) => {
    const file = path.join(RAM_DISK, req.params.canal, `${req.params.segmento}.ts`);
    if (fs.existsSync(file)) {
        // PERMITIR CACHÉ en los segmentos TS, ahorra peticiones al servidor si el usuario retrocede 
        res.setHeader('Cache-Control', 'public, max-age=3600');
        return res.sendFile(file);
    }
    res.status(404).end();
});

app.listen(PORT, () => {
    log(`⚡ Servidor TURBO iniciado en el puerto ${PORT} | Modo CPU 0%`);
});
