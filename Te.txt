const { spawn } = require('child_process');
const fs = require('fs');
const path = require('path');
const express = require('express');
const axios = require('axios');

const app = express();

const PORT = 8080;
const RAM_DISK = 'D:/HLS_LIVE'; // Carpeta temporal
const JSON_URL = 'https://raw.githubusercontent.com/boxplusteam/nose/refs/heads/main/data.json';

const INACTIVIDAD_MS = 30000; // 30 segundos

// Canales activos en memoria
const canalesActivos = new Map();

/*
Estructura ahora optimizada:
{
   id,
   proceso,
   folder,
   ultimaActividad: Date.now() // Timestamp de la última vez que alguien pidió un archivo
}
*/

// Asegurar carpeta
if (!fs.existsSync(RAM_DISK)) {
    try {
        fs.mkdirSync(RAM_DISK, { recursive: true });
    } catch (e) {
        console.log("❌ ERROR: No se puede escribir en D:");
    }
}

function log(...msg) {
    console.log(new Date().toLocaleTimeString(), ...msg);
}

function eliminarCanal(idCanal) {
    const canal = canalesActivos.get(idCanal);

    if (!canal) return;

    log(`🛑 Cerrando canal: ${idCanal}`);

    // Matar ffmpeg
    if (canal.proceso) {
        canal.proceso.kill('SIGKILL');
    }

    // Borrar archivos
    if (canal.folder && fs.existsSync(canal.folder)) {
        try {
            fs.rmSync(canal.folder, {
                recursive: true,
                force: true
            });
            log(`🗑️ Archivos eliminados: ${idCanal}`);
        } catch (err) {
            console.log("⚠️ Error borrando carpeta");
        }
    }

    canalesActivos.delete(idCanal);
}

// -------------------------------------------------------------
// NUEVO SISTEMA: Bucle global que revisa la inactividad real
// -------------------------------------------------------------
setInterval(() => {

        if (ahora - canal.ultimaActividad > INACTIVIDAD_MS) {
            log(`⌛ Canal inactivo por 30s. Cerrando automáticamente: ${id}`);
            eliminarCanal(id);
        }
    }
}, 10000); // Revisa todos los canales cada 10 segundos


async function iniciarCanal(idCanal) {

    // Si ya existe (ej. otro usuario entró a la vez), devolvemos el que ya está
    if (canalesActivos.has(idCanal)) {
        return canalesActivos.get(idCanal);
    }

    const { data } = await axios.get(JSON_URL);
    const canales = data.canales || data;
    const canalData = canales[idCanal];

    if (!canalData) {
        throw new Error('El canal no existe en el JSON');
    }

    const folderPath = path.join(RAM_DISK, idCanal);

    // Limpiar carpeta por seguridad antes de empezar
    if (fs.existsSync(folderPath)) {
        fs.rmSync(folderPath, { recursive: true, force: true });
    }
    fs.mkdirSync(folderPath, { recursive: true });

    log(`🚀 INICIANDO FFMPEG: ${idCanal}`);

    // Flags optimizados para evitar que FFmpeg se cuelgue si la fuente falla
    const ffmpeg = spawn('ffmpeg', [
        '-reconnect', '1',
        '-reconnect_streamed', '1',
        '-reconnect_at_eof', '1',
        '-reconnect_delay_max', '5', // Evita que se quede pillado intentando reconectar
        
        '-i', canalData.url,
        '-c', 'copy',
        '-f', 'hls',
        '-hls_time', '6',
        '-hls_list_size', '9',
        '-hls_flags', 'delete_segments+append_list', // Mejor manejo del disco
        
        path.join(folderPath, 'index.m3u8')
    ]);

    ffmpeg.stderr.on('data', (data) => {
        const txt = data.toString();
        // Filtramos para ver solo errores graves en consola y no saturarla
        if (txt.toLowerCase().includes('error') || txt.toLowerCase().includes('fatal')) {
            console.log(`[FFmpeg - ${idCanal}] ${txt.split('\n')[0]}`);
        }
    });

    ffmpeg.on('close', () => {
        log(`❌ FFmpeg cerrado o caído: ${idCanal}`);
        // Si se cae por un error externo, lo eliminamos de la lista para que 
        // el próximo usuario que entre fuerce un reinicio limpio
        if (canalesActivos.has(idCanal)) {
            eliminarCanal(idCanal);
        }
    });

    const canalObj = {
        id: idCanal,
        proceso: ffmpeg,
        folder: folderPath,
        ultimaActividad: Date.now() // Inicializamos el contador
    };

    canalesActivos.set(idCanal, canalObj);

    // -------------------------------------------------------------
    // OPTIMIZACIÓN: Esperar de forma inteligente al index.m3u8
    // -------------------------------------------------------------
    const m3u8Path = path.join(folderPath, 'index.m3u8');
    let intentos = 0;
    
    // Revisa cada medio segundo si el archivo ya existe (máximo 10 segundos)
    while (!fs.existsSync(m3u8Path) && intentos < 20) {
        await new Promise(r => setTimeout(r, 500));
        intentos++;
    }

    return canalObj;
}

// -------------------------------------------------------------
// RUTAS EXPRESS
// -------------------------------------------------------------

app.get('/:canal/index.m3u8', async (req, res) => {
    const idCanal = req.params.canal;

    try {
        let canal = canalesActivos.get(idCanal);

        // Iniciar el canal si no hay ningún FFmpeg haciéndolo
        if (!canal) {
            canal = await iniciarCanal(idCanal);
        } else {
            // ¡CLAVE! Si ya existe, actualizamos su actividad porque alguien lo está viendo
            canal.ultimaActividad = Date.now();
        }

        const file = path.join(RAM_DISK, idCanal, 'index.m3u8');

        if (fs.existsSync(file)) {
            return res.sendFile(file);
        }

        return res.status(503).send("Conectando con la fuente de video...");

    } catch (err) {
        console.log(`[ERROR ${idCanal}]`, err.message);
        return res.status(500).send(err.message);
    }
});

// Servir segmentos .ts
app.get('/:canal/:segmento.ts', (req, res) => {
    const idCanal = req.params.canal;
    const canal = canalesActivos.get(idCanal);

    // ¡CLAVE! Mientras el reproductor siga pidiendo trozos de video (.ts), 
    // mantenemos el canal vivo actualizando el timestamp.
    if (canal) {
        canal.ultimaActividad = Date.now();
    }

    const file = path.join(RAM_DISK, idCanal, `${req.params.segmento}.ts`);

    if (fs.existsSync(file)) {
        return res.sendFile(file);
    }

    return res.status(404).end();
});

// -------------------------------------------------------------
// INICIO DEL SERVIDOR
// -------------------------------------------------------------

app.listen(PORT, () => {
    console.log(`\n✅ Servidor HLS iniciado en el puerto ${PORT}`);
    console.log(`📡 Canales simultáneos permitidos: ILIMITADOS`);
    console.log(`⌛ Auto-cierre inteligente por inactividad: 30s`);
    console.log(`👥 Multiusuario: Si varias personas ven el mismo canal -> 1 solo FFmpeg\n`);
});
