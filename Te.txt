const { spawn } = require('child_process');
const fs = require('fs');
const path = require('path');
const express = require('express');
const axios = require('axios');

const app = express();

const PORT = 8080;
const RAM_DISK = 'D:/HLS_LIVE';
const JSON_URL = 'https://raw.githubusercontent.com/boxplusteam/nose/refs/heads/main/data.json';

const MAX_CANALES = 5;
const INACTIVIDAD_MS = 30000;

// canales activos
const canalesActivos = new Map();

/*
Estructura:

{
   id,
   proceso,
   folder,
   usuarios: Set(),
   timeout
}
*/

// asegurar carpeta
if (!fs.existsSync(RAM_DISK)) {
    try {
        fs.mkdirSync(RAM_DISK, { recursive: true });
    } catch (e) {
        console.log("❌ ERROR: No se puede escribir en D:");
    }
}

function log(...msg) {
    console.log(new Date().toLocaleTimeString(), ...msg);
}

function eliminarCanal(idCanal) {
    const canal = canalesActivos.get(idCanal);

    if (!canal) return;

    log(`🛑 Cerrando canal: ${idCanal}`);

    // matar ffmpeg
    if (canal.proceso) {
        canal.proceso.kill('SIGKILL');
    }

    // borrar timeout
    if (canal.timeout) {
        clearTimeout(canal.timeout);
    }

    // borrar archivos
    if (canal.folder && fs.existsSync(canal.folder)) {
        try {
            fs.rmSync(canal.folder, {
                recursive: true,
                force: true
            });

            log(`🗑️ Archivos eliminados: ${idCanal}`);
        } catch (err) {
            console.log("⚠️ Error borrando carpeta");
        }
    }

    canalesActivos.delete(idCanal);
}

function programarCierrePorInactividad(idCanal) {
    const canal = canalesActivos.get(idCanal);

    if (!canal) return;

    // limpiar timeout anterior
    if (canal.timeout) {
        clearTimeout(canal.timeout);
    }

    canal.timeout = setTimeout(() => {
        const actual = canalesActivos.get(idCanal);

        if (!actual) return;

        // si sigue vacío tras 30 segundos
        if (actual.usuarios.size === 0) {
            log(`⌛ Canal inactivo 30 segundos: ${idCanal}`);
            eliminarCanal(idCanal);
        }
    }, INACTIVIDAD_MS);
}

function usuarioSaleDelCanal(idCanal, userId) {
    const canal = canalesActivos.get(idCanal);

    if (!canal) return;

    canal.usuarios.delete(userId);

    log(`👤 Usuario salió de ${idCanal} | usuarios: ${canal.usuarios.size}`);

    // si no queda nadie
    if (canal.usuarios.size === 0) {
        programarCierrePorInactividad(idCanal);
    }
}

async function iniciarCanal(idCanal) {

    // si ya existe no iniciar otro ffmpeg
    if (canalesActivos.has(idCanal)) {
        return canalesActivos.get(idCanal);
    }

    // si ya hay 5 canales abiertos
    if (canalesActivos.size >= MAX_CANALES) {

        // buscar uno vacío
        for (const [id, canal] of canalesActivos) {

            if (canal.usuarios.size === 0) {
                log(`♻️ Liberando canal vacío: ${id}`);
                eliminarCanal(id);
                break;
            }
        }

        // si sigue habiendo 5 no dejar abrir más
        if (canalesActivos.size >= MAX_CANALES) {
            throw new Error('Limite de canales alcanzado');
        }
    }

    const { data } = await axios.get(JSON_URL);

    const canales = data.canales || data;

    const canalData = canales[idCanal];

    if (!canalData) {
        throw new Error('Canal no existe');
    }

    const folderPath = path.join(RAM_DISK, idCanal);

    // limpiar por seguridad
    if (fs.existsSync(folderPath)) {
        fs.rmSync(folderPath, {
            recursive: true,
            force: true
        });
    }

    fs.mkdirSync(folderPath, {
        recursive: true
    });

    log(`🚀 INICIANDO: ${idCanal}`);

    const ffmpeg = spawn('ffmpeg', [
        '-reconnect', '1',
        '-reconnect_streamed', '1',
        '-reconnect_at_eof', '1',

        '-i', canalData.url,

        '-c', 'copy',

        '-f', 'hls',

        '-hls_time', '6',
        '-hls_list_size', '9',

        '-hls_flags', 'delete_segments',

        path.join(folderPath, 'index.m3u8')
    ]);

    ffmpeg.stderr.on('data', (data) => {

        const txt = data.toString();

        if (txt.includes('Error')) {
            console.log(`[FFmpeg] ${txt.split('\n')[0]}`);
        }
    });

    ffmpeg.on('close', () => {
        log(`❌ FFmpeg cerrado: ${idCanal}`);
    });

    const canalObj = {
        id: idCanal,
        proceso: ffmpeg,
        folder: folderPath,
        usuarios: new Set(),
        timeout: null
    };

    canalesActivos.set(idCanal, canalObj);

    // esperar creación m3u8
    await new Promise(r => setTimeout(r, 4000));

    return canalObj;
}

/*
HEADER OBLIGATORIO:

x-user-id: ID_UNICO

Ejemplo:
x-user-id: pepe123
*/

app.get('/:canal/index.m3u8', async (req, res) => {

    const idCanal = req.params.canal;

    const userId =
        req.headers['x-user-id'] ||
        req.ip;

    try {

        let canal = canalesActivos.get(idCanal);

        // iniciar si no existe
        if (!canal) {
            canal = await iniciarCanal(idCanal);
        }

        // cancelar apagado si vuelve alguien
        if (canal.timeout) {
            clearTimeout(canal.timeout);
            canal.timeout = null;
        }

        // añadir usuario
        canal.usuarios.add(userId);

        log(`👤 Usuario dentro ${idCanal} | total: ${canal.usuarios.size}`);

        // detectar desconexión
        req.on('close', () => {
            usuarioSaleDelCanal(idCanal, userId);
        });

        const file = path.join(
            RAM_DISK,
            idCanal,
            'index.m3u8'
        );

        if (fs.existsSync(file)) {
            return res.sendFile(file);
        }

        return res
            .status(503)
            .send("Conectando con la fuente...");

    } catch (err) {

        console.log(err);

        return res
            .status(500)
            .send(err.message);
    }
});

// servir segmentos ts
app.get('/:canal/:segmento.ts', (req, res) => {

    const file = path.join(
        RAM_DISK,
        req.params.canal,
        `${req.params.segmento}.ts`
    );

    if (fs.existsSync(file)) {
        return res.sendFile(file);
    }

    return res.status(404).end();
});

app.listen(PORT, () => {

    console.log(`\n✅ Servidor iniciado puerto ${PORT}`);

    console.log(`📡 Máximo ${MAX_CANALES} canales simultáneos`);

    console.log(`⌛ Auto cierre por inactividad: 30s`);

    console.log(`👥 Si 2 usuarios ven el mismo canal -> 1 solo FFmpeg`);
});
